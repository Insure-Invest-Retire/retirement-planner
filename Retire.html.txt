  return pv*Math.pow(1+rm,n) + contribution*((Math.pow(1+rm,n)-1)/rm);
}
function requiredAtRetirement(v){
  const years = Math.max(1, v.life - v.retire);
  const monthlyExpense = v.expense*Math.pow(1+v.inflation, v.retire - v.age);
  const r = v.post/12, n = years*12;
  if(Math.abs(r)<1e-9) return monthlyExpense*n;
  return monthlyExpense*(1-Math.pow(1+r,-n))/r;
}

/* last computed result — ส่งไปพร้อมฟอร์มนัดหมาย */
let lastPlan = null;

function calculate(){
  const v = vals();
  const years = Math.max(1, v.retire - v.age);
  const projected = futureValue(v.current, years, v.pre, v.monthly);
  const target = requiredAtRetirement(v);
  const g = target - projected;
  const gap = Math.max(0, g);
  let need = 0;
  if(gap>0){
    const rm = v.pre/12, n = years*12;
    need = Math.abs(rm)<1e-9 ? gap/n : gap/((Math.pow(1+rm,n)-1)/rm);
  }
  // เงินลงทุน/เดือนที่ควรมี = ที่ลงอยู่ + ส่วนที่ขาด (ตัวเลขเดิมแสดงเฉพาะส่วนที่ต้องเพิ่ม)
  $('target').textContent = compact(target);
  $('projected').textContent = compact(projected);
  $('gap').textContent = g>0 ? '- '+compact(g) : '+ '+compact(-g);
  $('needed').textContent = money(need);
  $('heroTarget').textContent = compact(target);

  const badge = $('statusBadge');
  const statusText = g<=0 ? '● เกษียณได้ตามแผน' : g<target*.25 ? '● ใกล้ถึงเป้าหมาย' : '● ยังมี Retirement Gap';
  badge.className = 'status ' + (g<=0 ? '' : g<target*.25 ? 'warn' : 'bad');
  badge.textContent = statusText;
  $('expenseLabel').textContent = new Intl.NumberFormat('th-TH').format(v.expense);

  lastPlan = {
    age:v.age, retireAge:v.retire, lifeAge:v.life, currentSavings:v.current, monthlyInvest:v.monthly,
    monthlyExpense:v.expense, preReturnPct:v.pre*100, postReturnPct:v.post*100, inflationPct:v.inflation*100,
    target:Math.round(target), projected:Math.round(projected), gap:Math.round(g), neededPerMonth:Math.round(need),
    status:statusText.replace('● ','')
  };
  drawChart(v, target);
  drawScenarios(v);
}

function drawScenarios(v){
  const items = [['Conservative',Math.max(0,v.pre-.02)],['Base',v.pre],['Growth',v.pre+.02]];
  $('scenarios').innerHTML = items.map(([name,r])=>{
    const p = futureValue(v.current, v.retire-v.age, r, v.monthly);
    const t = requiredAtRetirement(v);
    return `<div class="scenario"><div class="rate">${name} • ${(r*100).toFixed(1)}%</div><b>${compact(p)}</b><div class="rate">${p>=t?'✓ เป้าหมายถึง':'Gap '+compact(t-p)}</div></div>`;
  }).join('');
}

function drawChart(v, target){
  const c = $('chart'), dpr = window.devicePixelRatio||1, rect = c.getBoundingClientRect();
  if(!rect.width) return;
  c.width = rect.width*dpr; c.height = 300*dpr;
  const ctx = c.getContext('2d'); ctx.setTransform(dpr,0,0,dpr,0,0);
  const W = rect.width, H = 300; ctx.clearRect(0,0,W,H);
  const start = v.age, end = Math.max(v.life, v.retire+1);

  // จำลองรายเดือน เก็บค่าปีละครั้ง
  const pts = [[start, v.current]];
  let bal = v.current;
  for(let a=start; a<end; a++){
    for(let m=0; m<12; m++){
      if(a < v.retire) bal = bal*(1+v.pre/12) + v.monthly;
      else bal = bal*(1+v.post/12) - v.expense*Math.pow(1+v.inflation, a-v.age);
    }
    pts.push([a+1, Math.max(0,bal)]);
  }
  const max = Math.max(target, ...pts.map(p=>p[1]), 1)*1.08;
  const X = a => 35 + (a-start)/(end-start)*(W-55);
  const Y = n => H-30 - n/max*(H-55);

  ctx.strokeStyle = '#e8e4db'; ctx.lineWidth = 1;
  for(let i=0;i<4;i++){ const y = 20+i*65; ctx.beginPath(); ctx.moveTo(35,y); ctx.lineTo(W-15,y); ctx.stroke(); }
  ctx.strokeStyle = '#c79a48'; ctx.setLineDash([5,5]);
  ctx.beginPath(); ctx.moveTo(X(v.retire),18); ctx.lineTo(X(v.retire),H-30); ctx.stroke(); ctx.setLineDash([]);
  ctx.strokeStyle = '#2e8066'; ctx.lineWidth = 3; ctx.beginPath();
  pts.forEach((p,i)=> i ? ctx.lineTo(X(p[0]),Y(p[1])) : ctx.moveTo(X(p[0]),Y(p[1]))); ctx.stroke();
  ctx.fillStyle = '#68746f'; ctx.font = '11px sans-serif';
  ctx.fillText('อายุ '+start, 35, H-8);
  ctx.fillText('เกษียณ '+v.retire, Math.min(X(v.retire)-25, W-110), H-8);
  ctx.fillText('อายุ '+end, W-55, H-8);
  ctx.fillStyle = '#c79a48'; ctx.fillText('Target '+compact(target), W-120, 25);
}

/* เบอร์ไทย: 0 + 8–9 หลัก  หรือ +66 / 66 นำหน้า */
function normalizePhone(raw){
  let d = String(raw).replace(/[^\d+]/g,'');
  if(d.startsWith('+66')) d = '0'+d.slice(3);
  else if(d.startsWith('66') && d.length>=11) d = '0'+d.slice(2);
  return d.replace(/\D/g,'');
}
const phoneOk = d => /^0\d{8,9}$/.test(d);
const emailOk = s => !s || /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(s);

function validate(){
  clearErrors();
  let bad = false;
  bad = setInvalid($('fFirst'), !$('fFirst').value.trim()) || bad;
  bad = setInvalid($('fLast'),  !$('fLast').value.trim())  || bad;
  bad = setInvalid($('fPhone'), !phoneOk(normalizePhone($('fPhone').value))) || bad;
  bad = setInvalid($('fEmail'), !emailOk($('fEmail').value.trim())) || bad;
  if(!$('fConsent').checked){ $('consentField').classList.add('invalid'); bad = true; }
  if(bad){ const first = form.querySelector('.invalid input, .invalid select'); if(first) first.focus(); }
  return !bad;
}

  }
